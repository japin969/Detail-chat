BAGIAN 2: DESIGN SYSTEM - LIQUID GLASS (WHATSAPP iOS 26)
=========================================
Prinsip: Transparan, blur background, rounded besar, highlight tepi, tanpa border tebal. Elemen glass "melayang" di atas wallpaper chat.

A. GLASS SURFACE (dipakai di TopBar, Composer, ContextMenu, Card):
   - Background: Color.White.copy(alpha = 0.72f) (light) / Color.Black.copy(alpha = 0.55f) (dark).
   - Blur (WAJIB pakai Haze):
     ```kotlin
     val hazeState = remember { HazeState() }
     // Di background layer (wallpaper):
     Modifier.hazeSource(state = hazeState)
     // Di glass surface:
     Modifier.hazeEffect(state = hazeState) {
         blurEffect {
             blurRadius = 20.dp
             colorEffects = listOf(HazeColorEffect.tint(Color.Black.copy(alpha = 0.1f)))
         }
     }
     ```
   - Shape: RoundedCornerShape(24.dp) untuk card, RoundedCornerShape(0.dp) untuk TopBar full-width.
   - Border (highlight tepi): Modifier.border(0.5.dp, Color.White.copy(alpha = 0.35f), shape).
   - Shadow: Modifier.shadow(8.dp, shape, ambientColor = Color.Black.copy(alpha = 0.15f), spotColor = Color.Black.copy(alpha = 0.20f)).

B. WALLPAPER CHAT (Background):
   - Base color: #EFEAE2 (light) / #0B141A (dark).
   - Tambahkan radial gradient overlay (Brush.radialGradient) dengan alpha rendah untuk kedalaman.
   - Seluruh konten chat (LazyColumn) menggunakan `Modifier.hazeSource(state = hazeState)` agar blur di glass surface mengambil dari wallpaper + bubble di belakangnya.

C. GLASS BUTTON (FAB, Send, Back):
   - CircleShape atau RoundedCornerShape(20.dp).
   - Brush.linearGradient(listOf(#25D366, #128C7E)) alpha 0.9.
   - Border 0.5.dp White alpha 0.4.
   - Pressed: alpha 0.7, scale 0.96.

D. GLASS CHIP (Filter, Reaction, Prompt):
   - RoundedCornerShape(16.dp).
   - Background glass alpha 0.55.
   - Border 0.5.dp White alpha 0.3.

package app.neon.com.data.model

import com.google.firebase.database.IgnoreExtraProperties

@IgnoreExtraProperties
data class User(
    val uid: String = "",
    val email: String = "",
    val username: String = "",
    val displayName: String = "",
    val avatarUrl: String = "",
    val avatarColorHex: String = "#00FF66",
    val bio: String = "Hey there! I am using Neont.",
    val followersCount: Int = 0,
    val followingCount: Int = 0,
    val likesCount: Int = 0,
    val showLastSeen: Boolean = true,
    val showProfilePhoto: Boolean = true,
    val createdAt: Long = System.currentTimeMillis(),
    val lastDisplayNameChange: Long = 0L,
    val lastUsernameChange: Long = 0L
) {
    val id: String get() = uid
    /**
     * Extracts 2-character initials for default avatar generation.
     */
    fun getInitials(): String {
        val cleanName = if (username.isNotBlank()) username else email.substringBefore("@")
        return when {
            cleanName.length >= 2 -> cleanName.take(2).uppercase()
            cleanName.length == 1 -> cleanName.take(1).uppercase()
            else -> "NE"
        }
    }
}

private fun avatarColorFor(name: String, colorHex: String?): Color {
    val parsed = colorHex?.takeIf { it.isNotBlank() }
        ?.let { hex -> runCatching { Color(android.graphics.Color.parseColor(hex)) }.getOrNull() }
    if (parsed != null) return parsed
    return FallbackAvatarColors[Math.floorMod(name.hashCode(), FallbackAvatarColors.size)]
}

private fun firstGlyph(word: String): String = String(Character.toChars(word.codePointAt(0)))

internal fun initialsOf(name: String): String {
    val words = name.trim().split(Regex("\\s+")).filter { it.isNotEmpty() }
    if (words.isEmpty()) return "?"
    return words.take(2).joinToString("") { firstGlyph(it) }.uppercase()
}

@OptIn(ExperimentalMaterial3Api::class)
@Composable
private fun SwipeableChatItem(
    chat: ChatItem,
    palette: NeontChatPalette,
    onClick: () -> Unit,
    onLongClick: () -> Unit,
    onTogglePin: () -> Unit,
    onDelete: () -> Unit
) {
    val dismissState = rememberSwipeToDismissBoxState(
        confirmValueChange = { value ->
            when (value) {
                SwipeToDismissBoxValue.StartToEnd -> {
                    onTogglePin()
                    false
                }
                SwipeToDismissBoxValue.EndToStart -> {
                    onDelete()
                    true
                }
                SwipeToDismissBoxValue.Settled -> false
            }
        }
    )

    SwipeToDismissBox(
        state = dismissState,
        enableDismissFromStartToEnd = true,
        enableDismissFromEndToStart = true,
        backgroundContent = {
            val direction = dismissState.dismissDirection
            val color by animateColorAsState(
                targetValue = when (dismissState.targetValue) {
                    SwipeToDismissBoxValue.StartToEnd -> IosBlue
                    SwipeToDismissBoxValue.EndToStart -> Color(0xFFFF3B30)
                    SwipeToDismissBoxValue.Settled -> Color.Transparent
                },
                label = "swipe_bg_color"
            )

            Box(
                modifier = Modifier
                    .fillMaxSize()
                    .background(color)
                    .padding(horizontal = 24.dp),
                contentAlignment = if (direction == SwipeToDismissBoxValue.StartToEnd) Alignment.CenterStart else Alignment.CenterEnd
            ) {
                if (direction == SwipeToDismissBoxValue.StartToEnd) {
                    Icon(Icons.Default.PushPin, contentDescription = "Pin", tint = Color.White)
                } else {
                    Icon(Icons.Default.Delete, contentDescription = "Hapus", tint = Color.White)
                }
            }
        }
    ) {
        ChatItemView(
            chat = chat,
            palette = palette,
            onClick = onClick,
            onLongClick = onLongClick
        )
    }
}

private fun formatTimeAgo(timestamp: Long): String {
    if (timestamp <= 0L) return ""
    val now = System.currentTimeMillis()
    val diff = now - timestamp
    return when {
        diff < 60_000L -> "baru saja"

/**
 * Gesture Geser untuk Membalas Pesan (Swipe-to-Reply)
 */
@Composable
private fun SwipeToReplyContainer(
    enabled: Boolean,
    onReply: () -> Unit,
    content: @Composable () -> Unit
) {
    val density = LocalDensity.current
    val thresholdPx = with(density) { 56.dp.toPx() }
    val maxPx = with(density) { 88.dp.toPx() }
    val offsetX = remember { Animatable(0f) }
    val scope = rememberCoroutineScope()
    val haptic = LocalHapticFeedback.current
    val latestOnReply by rememberUpdatedState(onReply)
    val palette = neontChatPalette()

    Box(
        modifier = Modifier
            .fillMaxWidth()
            .pointerInput(enabled) {
                if (!enabled) return@pointerInput
                var triggered = false
                var dragX = 0f
                detectHorizontalDragGestures(
                    onDragStart = {
                        triggered = false
                        dragX = offsetX.value
                    },
                    onDragEnd = { scope.launch { offsetX.animateTo(0f) } },
                    onDragCancel = { scope.launch { offsetX.animateTo(0f) } },
                    onHorizontalDrag = { change, dragAmount ->
                        change.consume()
                        dragX = (dragX + dragAmount).coerceIn(0f, maxPx)
                        scope.launch { offsetX.snapTo(dragX) }
                        if (!triggered && dragX >= thresholdPx) {
                            triggered = true
                            haptic.performHapticFeedback(HapticFeedbackType.LongPress)
                            latestOnReply()
                        }
                    }
                )
            }
    ) {
        Icon(
            imageVector = Icons.AutoMirrored.Filled.Reply,
            contentDescription = null,
            tint = palette.textSecondary,
            modifier = Modifier
                .align(Alignment.CenterStart)
                .padding(start = 16.dp)
                .size(20.dp)
                .graphicsLayer { alpha = (offsetX.value / thresholdPx).coerceIn(0f, 1f) }
        )
        Box(modifier = Modifier.offset { IntOffset(offsetX.value.roundToInt(), 0) }) {
            content()
        }
    }
}

private fun dayKey(timestamp: Long): Int {
    val calendar = Calendar.getInstance()
    calendar.timeInMillis = timestamp
    return calendar.get(Calendar.YEAR) * 1000 + calendar.get(Calendar.DAY_OF_YEAR)
}

private fun dayLabel(timestamp: Long): String {
    val now = System.currentTimeMillis()
    return when (dayKey(timestamp)) {
        dayKey(now) -> "Hari ini"
        dayKey(now - DAY_MS) -> "Kemarin"
        else -> SimpleDateFormat("d MMMM yyyy", Locale.forLanguageTag("id-ID")).format(Date(timestamp))
    }
}

private fun buildEntries(messages: List<ChatMessage>): List<ChatEntry> {
    val result = ArrayList<ChatEntry>(messages.size + 8)
    var lastDay = Int.MIN_VALUE
    messages.sortedBy { it.timestamp }.forEach { message ->
        if (message.timestamp > 0L) {
            val day = dayKey(message.timestamp)
            if (day != lastDay) {
                result.add(ChatEntry.Day(day, dayLabel(message.timestamp)))
                lastDay = day
            }
        }
        result.add(ChatEntry.Msg(message))
    }
    result.reverse()
    return result
}```

    clipboard.setPrimaryClip(ClipData.newPlainText("Neon Message", text))
    Toast.makeText(context, "Teks disalin (dihapus dalam 30 detik)", Toast.LENGTH_SHORT).show()

    }, 30_000L)
